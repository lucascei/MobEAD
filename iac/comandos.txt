# =============================================================================
# AZURE CLI — AUTENTICAÇÃO
# =============================================================================
az login

az account set --subscription "SUA_SUBSCRIPTION_ID"
az group list --output table

# Variáveis de ambiente (Service Principal - opcional)
export ARM_SUBSCRIPTION_ID="xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx"
export ARM_TENANT_ID="xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx"
export ARM_CLIENT_ID="xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx"
export ARM_CLIENT_SECRET="sua-chave-secreta"

# =============================================================================
# TERRAFORM
# =============================================================================
cd iac/terraform

# Configurar variáveis
cp terraform.tfvars.example terraform.tfvars

# Inicializar providers e módulos
terraform init

# Validar sintaxe
terraform validate

# Formatar código
terraform fmt -recursive

# Visualizar plano de execução
terraform plan -out=tfplan

# Aplicar infraestrutura
terraform apply tfplan

# Aplicar direto (sem salvar plano)
terraform apply -auto-approve

# Ver outputs (IP, nome VM, etc.)
terraform output
terraform output public_ip_address
terraform output vm_name
terraform output resource_group_name

# Destruir toda a infraestrutura
terraform destroy

# =============================================================================
# ANSIBLE — PRÉ-REQUISITOS
# =============================================================================
pip install ansible pywinrm
cd iac/ansible
ansible-galaxy collection install -r requirements.yml

# =============================================================================
# ANSIBLE — EXECUÇÃO
# =============================================================================
# Gerar inventory a partir do Terraform
./iac/scripts/generate-inventory.sh

# Testar conexão WinRM
cd iac/ansible
ansible windows -i inventory.ini -m ansible.windows.win_ping

# Executar playbook completo
ansible-playbook -i inventory.ini playbook.yml

# Executar apenas validação
ansible-playbook -i inventory.ini playbook.yml --tags validate
